import gradio as gr
import whisper
import torch
from transformers import DistilBertTokenizer, DistilBertForSequenceClassification
import numpy as np
import time

# Load Whisper model for speech-to-text
whisper_model = whisper.load_model("base", device="cpu")  # Use "tiny" for faster processing if needed

# Load DistilBERT model and tokenizer for spam detection
model_name = "AventIQ-AI/distilbert-spam-detection"  # Pretrained spam detection model
tokenizer = DistilBertTokenizer.from_pretrained(model_name)
spam_model = DistilBertForSequenceClassification.from_pretrained(model_name)
device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
spam_model.to(device)

# Function to transcribe audio using Whisper
def transcribe_audio(audio):
    if audio is None:
        return "No audio input provided."
    # Whisper expects audio as a NumPy array with sample rate 16000
    result = whisper_model.transcribe(audio, fp16=False)
    return result["text"]

# Function to predict if text is spam
def predict_spam(text):
    spam_model.eval()
    inputs = tokenizer(text, return_tensors="pt", padding="max_length", truncation=True, max_length=128).to(device)
    
    with torch.no_grad():
        outputs = spam_model(**inputs)
        probs = torch.softmax(outputs.logits, dim=-1)
        pred_class = torch.argmax(probs).item()
    
    return "🚨 Potential Scam Detected!" if pred_class == 1 else "✅ Safe"

# Combined function for Gradio: Audio -> Transcription -> Prediction
def process_audio(audio):
    # Transcribe the audio
    transcription = transcribe_audio(audio)
    # Predict if it's spam
    prediction = predict_spam(transcription)
    return transcription, prediction

# Gradio Interface
with gr.Blocks(theme=gr.themes.Soft()) as demo:
    gr.Markdown("# 🚀 Real-Time Audio Scam Detector")
    gr.Markdown("Record or upload audio to detect potential scams in real-time (simulated for demo).")
    
    with gr.Row():
        audio_input = gr.Audio(sources=["microphone", "upload"], type="filepath", label="Record or Upload Audio")
        transcribe_output = gr.Textbox(label="Transcription", placeholder="Transcription will appear here...")
        prediction_output = gr.Label(label="Prediction")
    
    button = gr.Button("🔍 Analyze Audio")
    button.click(process_audio, inputs=audio_input, outputs=[transcribe_output, prediction_output])

# Launch the demo
if __name__ == "__main__":
    demo.launch()
